**Issue:**

            fn fmt(&self, f: &mut ::std::fmt::Formatter<'_>) -> ::std::fmt::Result {
                write!(f, concat!(stringify!($name), "-{}"), self.0)
            }
        }

        impl $crate::AggregateId for $name {
            fn to_uuid(&self) -> ::uuid::Uuid {
                self.0
            }
        }

### 2. Repository Pattern 🔵 **NICE TO HAVE**

**Purpose:** Higher-level abstraction over `EventStore`.

**Implementation:**
```rust
// crates/event-sauce-core/src/repository.rs

use std::marker::PhantomData;
use std::sync::Arc;

/// Repository provides a high-level API for aggregate persistence.
///
/// Wraps an EventStore and provides type-safe operations for a specific
/// aggregate type.
///
/// # Examples
///
/// ```ignore
/// let repo = Repository::<PostgresEventStore, User>::new(store);
///
/// // Load aggregate
/// let user = repo.load(user_id).await?;
///
/// // Save aggregate
/// repo.save(&mut user).await?;
///
/// // Check existence
/// if repo.exists(user_id).await? {
///     // ...
/// }
/// ```
pub struct Repository<S, A> {
    store: Arc<S>,
    _phantom: PhantomData<A>,
}

impl<S, A> Repository<S, A>
where
    S: EventStore + 'static,
    A: Aggregate + Serialize,
    A::Event: Serialize + DeserializeOwned,
{
    /// Creates a new repository wrapping the given event store.
    pub fn new(store: Arc<S>) -> Self {
        Self {
            store,
            _phantom: PhantomData,
        }
    }

    /// Saves an aggregate to the event store.
    ///
    /// Commits all pending events and clears them on success.
    pub async fn save(&self, aggregate: &mut A) -> Result<()> {
        self.store.commit(aggregate).await
    }

    /// Loads an aggregate from the event store.
    ///
    /// Reconstructs the aggregate by replaying all its events.
    pub async fn load(&self, id: A::Id) -> Result<A> {
        load(&*self.store, id).await
    }

    /// Checks if an aggregate exists.
    ///
    /// Returns true if the aggregate has any events in the store.
    pub async fn exists(&self, id: A::Id) -> Result<bool> {
        let stream_id = StreamId::new(A::aggregate_type(), id.to_uuid());
        self.store.stream_exists(stream_id).await
    }

    /// Gets the current version of an aggregate without loading it.
    pub async fn get_version(&self, id: A::Id) -> Result<Version> {
        let stream_id = StreamId::new(A::aggregate_type(), id.to_uuid());
        self.store.get_version(stream_id).await
    }

    /// Counts events for an aggregate.
    pub async fn count_events(&self, id: A::Id) -> Result<usize> {
        let stream_id = StreamId::new(A::aggregate_type(), id.to_uuid());
        count_events(&*self.store, stream_id).await
    }
}

impl<S, A> Clone for Repository<S, A> {
    fn clone(&self) -> Self {
        Self {
            store: Arc::clone(&self.store),
            _phantom: PhantomData,
        }
    }
}
```

**Benefits:**
- ✅ Type-safe aggregate operations
- ✅ Cleaner API than raw EventStore
- ✅ Easier to mock for testing
- ✅ Domain-focused interface

**Usage Example:**
```rust
// Before:

let mut user: User = load(&event_store, user_id).await?;
user.verify_email()?;
event_store.commit(&mut user).await?;

// After:

let mut user = user_repo.load(user_id).await?;
user.verify_email()?;
user_repo.save(&mut user).await?;

### 3. Projection Builder Macro 🔵 **NICE TO HAVE**

**Purpose:** Simplify creation of read models.

**Implementation:**
```rust
#[macro_export]
macro_rules! projection {
    (
        $vis:vis struct $name:ident {
            state: $state:ty,

            $(
                on $event:ty => |$proj:ident, $evt:ident| $handler:block
            ),* $(,)?
        }
    ) => {
        $vis struct $name {
            state: $state,
        }

        impl $name {
            pub fn new(state: $state) -> Self {
                Self { state }
            }

            pub fn state(&self) -> &$state {
                &self.state
            }

            pub fn state_mut(&mut self) -> &mut $state {
                &mut self.state
            }

            pub async fn handle(&mut self, event: &EventEnvelope)
                -> $crate::Result<()>
            {
                $(
                    if let Ok($evt) = <$event>::from_envelope(event) {
                        let $proj = self;
                        $handler
                        return Ok(());
                    }
                )*
                Ok(()) // Ignore unknown events
            }
        }
    };
}

// Usage:
projection! {
    pub struct UserListProjection {
        state: HashMap<UserId, UserView>,

        on UserRegisteredEvent => |proj, event| {
            proj.state.insert(event.user_id, UserView {
                email: event.email.clone(),
                status: UserStatus::Active,
                registered_at: event.timestamp,
            });
        },

        on UserEmailChangedEvent => |proj, event| {
            if let Some(user) = proj.state.get_mut(&event.user_id) {
                user.email = event.new_email.clone();
            }
        },

        on UserDeletedEvent => |proj, event| {
            proj.state.remove(&event.user_id);
        },
    }
}
```

**Benefits:**
- ✅ Declarative projection definition
- ✅ Type-safe event handling
- ✅ Automatic event deserialization

### 4. Event Definition Macro 🔵 **NICE TO HAVE**

**Purpose:** Reduce boilerplate for simple events.

**Current Pattern:**
```rust
#[derive(Debug, Clone, Serialize, Deserialize)]
struct CounterIncrementedEvent {
    amount: i32,
    timestamp: DateTime<Utc>,
}

impl ApplyEvent<Counter> for CounterIncrementedEvent {
    fn apply(&self, counter: &mut Counter) {
        counter.value += self.amount;
    }
}

// Need to wrap in enum, implement DomainEvent, etc.
```

**Proposed Macro:**
```rust
#[macro_export]
macro_rules! define_events {
    (
        $vis:vis enum $name:ident for $aggregate:ty {
            $(
                $variant:ident {
                    $($field:ident: $ty:ty),* $(,)?
                } => $apply:expr
            ),* $(,)?
        }
    ) => {
        #[derive(Debug, Clone, ::serde::Serialize, ::serde::Deserialize)]
        $vis enum $name {
            $(
                $variant {
                    $($field: $ty,)*
                    timestamp: ::chrono::DateTime<::chrono::Utc>,
                },
            )*
        }

        impl $crate::DomainEvent for $name {
            type Aggregate = $aggregate;

            fn event_type(&self) -> &'static str {
                match self {
                    $(Self::$variant { .. } => stringify!($variant),)*
                }
            }

            fn event_version(&self) -> i32 { 1 }

            fn occurred_at(&self) -> ::chrono::DateTime<::chrono::Utc> {
                match self {
                    $(Self::$variant { timestamp, .. } => *timestamp,)*
                }
            }
        }

        // Generate ApplyEvent for each variant
        $(
            paste::paste! {
                #[derive(Debug, Clone, ::serde::Serialize, ::serde::Deserialize)]
                $vis struct [<$variant Event>] {
                    $($field: $ty,)*
                    timestamp: ::chrono::DateTime<::chrono::Utc>,
                }

                impl $crate::ApplyEvent<$aggregate> for [<$variant Event>] {
                    fn apply(&self, aggregate: &mut $aggregate) {
                        let apply_fn: fn(&mut $aggregate, &Self) = $apply;
                        apply_fn(aggregate, self);
                    }
                }

                impl Into<$name> for [<$variant Event>] {
                    fn into(self) -> $name {
                        $name::$variant {
                            $($field: self.$field,)*
                            timestamp: self.timestamp,
                        }
                    }
                }
            }
        )*
    };
}

// Usage:
define_events! {
    pub enum CounterEvent for Counter {
        Incremented {
            amount: i32
        } => |counter, event| {
            counter.value += event.amount;
        },

        Decremented {
            amount: i32
        } => |counter, event| {
            counter.value -= event.amount;
        },

        Reset {} => |counter, _| {
            counter.value = 0;
        },
    }
}
```

**Benefits:**
- ✅ 60% less boilerplate
- ✅ Automatic timestamp handling
- ✅ Automatic event structs

### 5. Aggregate Root Helper 🔵 **NICE TO HAVE**

**Purpose:** Default implementation for simple aggregates.

**Implementation:**
```rust
// crates/event-sauce-core/src/aggregate_root.rs

/// Generic aggregate root implementation for simple aggregates.
///
/// Provides default implementations of the Aggregate trait for aggregates
/// that don't need custom behavior.
///
/// # Examples
///
/// ```ignore
/// type Counter = AggregateRoot<CounterId, CounterEvent, CounterState, CounterError>;
///
/// let mut counter = Counter::new(CounterId::new());
/// counter.apply(CounterEvent::Incremented { amount: 5 })?;
/// ```
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct AggregateRoot<ID, E, S, Err = ()> {
    pub id: ID,
    pub state: S,
    pub version: Version,
    pub pending_events: Vec<E>,
    _error: PhantomData<Err>,
}

impl<ID, E, S, Err> AggregateRoot<ID, E, S, Err>
where
    ID: AggregateId,
    E: DomainEvent,
    S: Default + Serialize + DeserializeOwned + Send + Sync,
    Err: AggregateError,
{
    pub fn new(id: ID) -> Self {
        Self {
            id,
            state: S::default(),
            version: Version::initial(),
            pending_events: Vec::new(),
            _error: PhantomData,
        }
    }
}

impl<ID, E, S, Err> Aggregate for AggregateRoot<ID, E, S, Err>
where
    ID: AggregateId,
    E: DomainEvent<Aggregate = Self> + ApplyEvent<Self>,
    S: Default + Serialize + DeserializeOwned + Send + Sync,
    Err: AggregateError,
{
    type Event = E;
    type Id = ID;
    type Error = Err;
    type State = S;

    fn new(id: Self::Id) -> Self {
        Self::new(id)
    }

    fn aggregate_id(&self) -> &Self::Id {
        &self.id
    }

    fn version(&self) -> Version {
        self.version
    }

    fn pending_events(&self) -> &[Self::Event] {
        &self.pending_events
    }

    fn clear_pending_events(&mut self) {
        self.pending_events.clear();
    }

    fn apply<Ev: Into<Self::Event>>(&mut self, event: Ev)
        -> Result<(), Self::Error>
    {
        let event = event.into();
        self.apply_internal(&event)?;
        self.pending_events.push(event);
        Ok(())
    }

    fn apply_internal(&mut self, event: &Self::Event)
        -> Result<(), Self::Error>
    {
        event.validate(self)?;
        event.apply(self);
        event.post_validate(self)?;
        self.version = self.version.next();
        Ok(())
    }

    fn state(&self) -> &Self::State {
        &self.state
    }

    fn from_snapshot(id: Self::Id, version: Version, state: Self::State) -> Self {
        Self {
            id,
            state,
            version,
            pending_events: Vec::new(),
            _error: PhantomData,
        }
    }
}
```
